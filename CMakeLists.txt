cmake_minimum_required(VERSION 3.22)
project(VocaBox VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(VOCA_DSP_ONLY "Build DSP tests without JUCE or GUI dependencies" OFF)
add_executable(vocabox_dsp_tests Tests/DspTests.cpp)
add_test(NAME DSP COMMAND vocabox_dsp_tests)
add_executable(vocabox_tube_tests Tests/TubeTests.cpp)
add_test(NAME Tube COMMAND vocabox_tube_tests)
if(VOCA_DSP_ONLY)
 return()
endif()
if(DEFINED JUCE_SOURCE_DIR)
 add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/JUCE")
else()
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 4f43011b96eb0636104cb3e433894cda98243626 GIT_SHALLOW FALSE)
 FetchContent_MakeAvailable(JUCE)
 set(JUCE_SOURCE_DIR "${juce_SOURCE_DIR}")
endif()
include(cmake/CompatibleGraphics.cmake)
juce_add_binary_data(VocaResources SOURCES Resources/face.png Resources/presets.json Resources/knobs.png Resources/InterSemiBold.ttf)
juce_add_plugin(VocaBox
 COMPANY_NAME "VocaBox Project" BUNDLE_ID "org.vocabox.channelstrip"
 PLUGIN_MANUFACTURER_CODE VcBx PLUGIN_CODE Vb01
 FORMATS VST3 PRODUCT_NAME "VocaBox"
 IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
 VST3_CATEGORIES Fx Dynamics EQ VST3_AUTO_MANIFEST FALSE
 COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(VocaBox PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/NavegadorPresets.cpp)
target_compile_options(VocaBox PUBLIC $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
target_compile_definitions(VocaBox PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_compile_definitions(VocaBox PUBLIC CHI_WINDOWS_SOFTWARE_GRAPHICS=$<BOOL:${CHI_WINDOWS_SOFTWARE_GRAPHICS}>)
if(WIN32)
 target_compile_definitions(VocaBox PUBLIC JUCE_DIRECT2D=1)
endif()
target_link_libraries(VocaBox PRIVATE VocaResources juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
if(MINGW)
 target_link_options(VocaBox_VST3 PRIVATE -static-libgcc -static-libstdc++ -static)
endif()
if(BUILD_TESTING)
 add_executable(vocabox_processor_tests Tests/ProcessorTests.cpp)
 target_link_libraries(vocabox_processor_tests PRIVATE VocaBox VocaResources)
 target_include_directories(vocabox_processor_tests PRIVATE $<TARGET_PROPERTY:VocaBox,INCLUDE_DIRECTORIES>)
 target_compile_definitions(vocabox_processor_tests PRIVATE $<TARGET_PROPERTY:VocaBox,COMPILE_DEFINITIONS>)
 if(MINGW)
  target_link_options(vocabox_processor_tests PRIVATE -static-libgcc -static-libstdc++ -static)
 endif()
 if(NOT CMAKE_CROSSCOMPILING)
  add_test(NAME Processor COMMAND vocabox_processor_tests)
 endif()
endif()

if(BUILD_TESTING)
 add_executable(vocabox_vst3_smoke Tests/Vst3HostSmoke.cpp)
 target_include_directories(vocabox_vst3_smoke PRIVATE "${JUCE_SOURCE_DIR}/modules/juce_audio_processors/format_types/VST3_SDK")
 if(UNIX)
  target_link_libraries(vocabox_vst3_smoke PRIVATE ${CMAKE_DL_LIBS})
 endif()
 if(MINGW)
  target_link_options(vocabox_vst3_smoke PRIVATE -static-libgcc -static-libstdc++ -static)
 endif()
 add_dependencies(vocabox_vst3_smoke VocaBox_VST3)
 if(NOT CMAKE_CROSSCOMPILING)
  add_test(NAME VST3Host COMMAND vocabox_vst3_smoke $<TARGET_FILE:VocaBox_VST3>)
 endif()
endif()

if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
 add_executable(vocabox_editor_lifetime Tests/EditorLifetimeTests.cpp)
 target_link_libraries(vocabox_editor_lifetime PRIVATE VocaBox VocaResources)
 target_include_directories(vocabox_editor_lifetime PRIVATE $<TARGET_PROPERTY:VocaBox,INCLUDE_DIRECTORIES>)
 target_compile_definitions(vocabox_editor_lifetime PRIVATE $<TARGET_PROPERTY:VocaBox,COMPILE_DEFINITIONS>)
 add_test(NAME EditorLifetime COMMAND vocabox_editor_lifetime)
 set_tests_properties(EditorLifetime PROPERTIES TIMEOUT 130)
endif()
